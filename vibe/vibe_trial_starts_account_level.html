-- vibe_trial_starts_account_level.sql
-- Run daily. Replace DATA in the HTML with query results.
-- Excludes monday.com internal accounts.
-- 3 segments: Touch Wire (Free Trial), Touch CC (Autopurchase), No Touch (Autopurchase)

SELECT
  dt,
  segment,             -- 'Touch Wire', 'Touch CC', or 'No Touch'
  trial_method,        -- 'Free Trial' or 'Autopurchase'
  COUNT(DISTINCT account_id) AS trial_starts,
  vibe_accounts,
  ROUND(100.0 * COUNT(DISTINCT account_id) / NULLIF(vibe_accounts, 0), 2) AS cvr_pct
FROM (
  SELECT
    DATE(t.created_at) AS dt,
    t.account_id,
    CASE
      WHEN t.is_touch = 1 AND t.trial_type != 'autopurchase' THEN 'Touch Wire'
      WHEN t.is_touch = 1 AND t.trial_type = 'autopurchase'  THEN 'Touch CC'
      ELSE 'No Touch'
    END AS segment,
    CASE WHEN t.trial_type = 'autopurchase' THEN 'Autopurchase' ELSE 'Free Trial' END AS trial_method
  FROM bigbrain.final.trial_starts t
  LEFT JOIN bigbrain.final.daccounts a ON t.account_id = a.account_id
  WHERE t.product = 'vibe'
    AND t.created_at >= '2025-02-01'
    AND (a.is_monday_account = 0 OR a.is_monday_account IS NULL)  -- exclude monday accounts
    AND NOT (t.is_touch = 0 AND t.trial_type != 'autopurchase')   -- no free trial for no-touch
) trials
LEFT JOIN (
  SELECT DATE(event_time) AS dt, COUNT(DISTINCT v.account_id) AS vibe_accounts
  FROM bigbrain.final.vibe_events v
  LEFT JOIN bigbrain.final.daccounts a ON v.account_id = a.account_id
  WHERE v.event_time >= '2025-02-01'
    AND (a.is_monday_account = 0 OR a.is_monday_account IS NULL)  -- exclude monday accounts
  GROUP BY 1
) vibe USING (dt)
GROUP BY dt, segment, trial_method, vibe_accounts
ORDER BY dt, segment;
