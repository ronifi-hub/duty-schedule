-- vibe_publish_by_interaction_type.sql
-- Non-paying Vibe accounts: publish probability by first interaction type
-- Also: 2nd prompt rate and its effect on publish
-- Segments: Discuss Mode | Prompt Only | Template
-- Verify event names before running!

WITH params AS (
  SELECT DATE('{{date_range.start}}') AS start_date,
         DATE('{{date_range.end}}')   AS end_date
),

-- Non-paying accounts: no active Vibe subscription or trial
non_paying AS (
  SELECT DISTINCT a.pulse_account_id
  FROM bigbrain.final.daccounts a
  WHERE (a.is_monday_account = 0 OR a.is_monday_account IS NULL)
    -- Adjust these filters based on your actual schema for "non-paying":
    -- Option A: no Vibe trial ever started
    AND NOT EXISTS (
      SELECT 1 FROM bigbrain.final.trial_starts t
      WHERE t.account_id = a.account_id AND t.product = 'vibe'
    )
    -- Option B: check plan_id / subscription status
    -- AND (a.vibe_plan_id IS NULL OR a.vibe_plan_id = 'free')
),

admin_users AS (
  SELECT DISTINCT pulse_account_id, pulse_user_id
  FROM bigbrain.final.dusers
  WHERE is_admin = 1 AND enabled = 1
),

-- All relevant Vibe events for non-paying admin users
ev AS (
  SELECT e.pulse_account_id, e.pulse_user_id, e.name, e.created_at
  FROM bigbrain.final.events e
  INNER JOIN non_paying np ON np.pulse_account_id = e.pulse_account_id
  INNER JOIN admin_users au ON au.pulse_account_id = e.pulse_account_id
                            AND au.pulse_user_id = e.pulse_user_id
  CROSS JOIN params p
  WHERE DATE(e.created_at) BETWEEN p.start_date AND p.end_date
    AND e.name IN (
      'ai_app_chat_mode_clicked',        -- Discuss mode
      'ai_app_user_prompt_received',     -- Prompt sent
      'ai_app_template_clicked',         -- Template used (VERIFY event name)
      'ai_app_preview_rendered',         -- Preview rendered
      'ai_app_publish_clicked'           -- Publish clicked
    )
),

-- Classify each user by first interaction type
user_first_action AS (
  SELECT pulse_user_id, segment
  FROM (
    SELECT pulse_user_id,
      CASE
        WHEN name = 'ai_app_chat_mode_clicked' THEN 'Discuss Mode'
        WHEN name = 'ai_app_template_clicked'  THEN 'Template'
        ELSE 'Prompt Only'
      END AS segment,
      ROW_NUMBER() OVER (PARTITION BY pulse_user_id ORDER BY created_at) AS rn
    FROM ev
    WHERE name IN (
      'ai_app_chat_mode_clicked',
      'ai_app_user_prompt_received',
      'ai_app_template_clicked'
    )
  ) t
  WHERE rn = 1
),

-- Prompt count per user
user_prompt_count AS (
  SELECT pulse_user_id, COUNT(*) AS prompt_count
  FROM ev
  WHERE name = 'ai_app_user_prompt_received'
  GROUP BY pulse_user_id
),

-- Users who saw preview
user_preview AS (
  SELECT DISTINCT pulse_user_id
  FROM ev
  WHERE name = 'ai_app_preview_rendered'
),

-- Users who clicked publish
user_publish AS (
  SELECT DISTINCT pulse_user_id
  FROM ev
  WHERE name = 'ai_app_publish_clicked'
),

-- Main result set
user_summary AS (
  SELECT
    fa.pulse_user_id,
    fa.segment,
    COALESCE(pc.prompt_count, 0) AS prompt_count,
    CASE WHEN pr.pulse_user_id IS NOT NULL THEN 1 ELSE 0 END AS saw_preview,
    CASE WHEN pu.pulse_user_id IS NOT NULL THEN 1 ELSE 0 END AS clicked_publish
  FROM user_first_action fa
  LEFT JOIN user_prompt_count pc ON pc.pulse_user_id = fa.pulse_user_id
  LEFT JOIN user_preview pr      ON pr.pulse_user_id = fa.pulse_user_id
  LEFT JOIN user_publish pu      ON pu.pulse_user_id = fa.pulse_user_id
)

-- Query 1: Publish rate by segment
SELECT segment,
       COUNT(*) AS total_users,
       SUM(saw_preview) AS saw_preview,
       SUM(clicked_publish) AS clicked_publish,
       ROUND(100.0 * SUM(clicked_publish) / NULLIF(SUM(saw_preview), 0), 1) AS publish_rate_pct
FROM user_summary
WHERE saw_preview = 1
GROUP BY segment
ORDER BY segment;
